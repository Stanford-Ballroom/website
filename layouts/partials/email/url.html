{{- /* Links and exported images use the public origin; local preview images stay local. */ -}} {{- $url := strings.TrimSpace .url -}} {{- $base := strings.TrimSuffix "/" site.Data.emails.base_url -}} {{- $previewImage := and .image (ne hugo.Environment "email") (or hugo.IsServer (eq hugo.Environment "development")) (hasPrefix $url "/") (not (hasPrefix $url "//")) -}} {{- /* Hugo-generated permalinks may contain a preview deployment subdirectory. */ -}} {{- if and .generated (not $previewImage) -}} {{- $prefix := strings.TrimSuffix "/" (urls.Parse site.BaseURL).Path -}} {{- if $prefix }}{{ $url = strings.TrimPrefix $prefix $url }}{{ end -}} {{- end -}} {{- if not (hasPrefix $base "https://") -}} {{- errorf "data/emails.yaml: base_url must be an absolute HTTPS URL" -}} {{- end -}} {{- if not $url -}} {{- errorf "Empty email URL in %s" .page.Path -}} {{- else if hasPrefix $url "//" -}} {{- $url = print "https:" $url -}} {{- else if hasPrefix $url "/" -}} {{- if $previewImage -}} {{- if not .generated }}{{ $url = strings.TrimPrefix "/" $url | relURL }}{{ end -}} {{- else -}} {{- $url = print $base $url -}} {{- end -}} {{- else if hasPrefix $url "#" -}} {{- $pageURL := partial "email/url.html" (dict "url" .page.RelPermalink "page" .page "generated" true) -}} {{- $url = print $pageURL $url -}} {{- else if not (findRE `(?i)^(https?://|mailto:|tel:)` $url) -}} {{- errorf "Email URL %q in %s must be absolute or start with / (site root)" $url .page.Path -}} {{- end -}} {{- return $url -}}
